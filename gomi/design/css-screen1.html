画面1 のCSS（コード）
/* 全体：背景・文字・書体 */
body {
margin: 0;← 外側の余白をなくす
background: #eef2ee;← 背景の色（うすい緑のグレー）
color: #18221d;← 文字の色（ほぼ黒）
font-family: "BIZ UDPGothic", sans-serif;← 書体
line-height: 1.7;← 行の高さ（文字の1.7倍）
}
/* 白い画面 */
main {
max-width: 420px;← 横幅は最大420px
min-height: 100vh;← 高さは最低でも画面いっぱい
margin: 0 auto;← 左右の余白を自動にして真ん中に置く
padding: 16px;← 内側の余白16px
box-sizing: border-box;← 余白と枠も含めて幅を計算する
background: #fff;← 背景を白にする
}
/* ステップ表示 */
.step {
color: #4c5952;← 文字を薄い色にする
font-size: 14px;← 文字の大きさ14px
}
/* 見出し */
h1 {
font-size: 22px;← 文字の大きさ22px
margin-top: 0;← 上の余白をなくす
}
/* 選択肢の上の見出し */
.group-title {
font-weight: bold;← 太字にする
margin-bottom: 8px;← 下に8px空ける
}
/* 選択肢のカード */
.card {
display: flex;← 中身（チェックと文字）を横に並べる
gap: 10px;← 並べた中身の間を10px空ける
padding: 12px;← 内側の余白12px
margin-bottom: 8px;← 下に8px空ける（カードどうしの間）
border: 2px solid #c3cdc6;← 2pxの薄いグレーの枠線
border-radius: 8px;← 角を8px丸くする
}
/* 選んだカード */
.card:has(input:checked) {
border-color: #1d6a4c;← 枠線の色を緑にする
background: #dcebe2;← 背景を薄い緑にする
}
/* 手数料の合計 */
.total {
font-size: 18px;← 文字の大きさ18px
font-weight: bold;← 太字にする
}
/* エラー文 */
.error {
color: #b42318;← 文字を赤にする
font-weight: bold;← 太字にする
margin: 4px 0;← 上下に4px空ける（左右は0）
}
/* ボタン */
button {
width: 100%;← 横幅いっぱいにする
min-height: 48px;← 高さは最低48px（指で押しやすい）
margin-top: 8px;← 上に8px空ける
font-size: 16px;← 文字の大きさ16px
font-weight: bold;← 太字にする
border: 2px solid #1d6a4c;← 2pxの緑の枠線
border-radius: 8px;← 角を8px丸くする
background: #1d6a4c;← 背景を緑にする
color: #fff;← 文字を白にする
}
